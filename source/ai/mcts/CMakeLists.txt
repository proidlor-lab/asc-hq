# MCTS AI Module
# Modern C++23 with strict warnings

message(STATUS "Configuring MCTS module...")

# Find SDL and sigc++ (required for core ASC types)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL REQUIRED sdl)
pkg_check_modules(SIGC REQUIRED sigc++-2.0)

# Auto-discover source files by category
file(GLOB DOMAIN_SOURCES "domain/*.cpp")
file(GLOB DOMAIN_ABILITY_SOURCES "domain/abilities/*.cpp")
file(GLOB AGENT_SOURCES "agents/*.cpp")
file(GLOB CORE_SOURCES "core/*.cpp")
file(GLOB INFRA_SOURCES "infrastructure/*.cpp")

# Exclude test files from library sources
list(FILTER DOMAIN_SOURCES EXCLUDE REGEX ".*_test\\.cpp$")
list(FILTER AGENT_SOURCES EXCLUDE REGEX ".*_test\\.cpp$")
list(FILTER CORE_SOURCES EXCLUDE REGEX ".*_test\\.cpp$")
list(FILTER INFRA_SOURCES EXCLUDE REGEX ".*_test\\.cpp$")

# Add manual test file explicitly
set(MANUAL_TEST_SOURCE "")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/mcts_manual_test.cpp")
    set(MANUAL_TEST_SOURCE mcts_manual_test.cpp)
endif()

# Build MCTS library
add_library(mcts
    ${DOMAIN_SOURCES}
    ${DOMAIN_ABILITY_SOURCES}
    ${AGENT_SOURCES}
    ${CORE_SOURCES}
    ${INFRA_SOURCES}
    ${MANUAL_TEST_SOURCE}
)

# Force C++23 for MCTS module (it uses modern features)
set_target_properties(mcts PROPERTIES
    CXX_STANDARD 23
    CXX_STANDARD_REQUIRED ON
)

# Include directories
target_include_directories(mcts PUBLIC
    ${CMAKE_SOURCE_DIR}/source
    ${CMAKE_SOURCE_DIR}  # For config.h (generated by Autotools during transition)
    ${CMAKE_SOURCE_DIR}/source/libs/loki-0.1.6/include
    ${CMAKE_CURRENT_SOURCE_DIR}  # For relative includes within MCTS module (e.g., "core/mcts_config.h")
    ${SDL_INCLUDE_DIRS}
    ${SIGC_INCLUDE_DIRS}
)

# Optional: Add paragui include if it exists
if(EXISTS "${CMAKE_SOURCE_DIR}/source/libs/paragui/include")
    target_include_directories(mcts PUBLIC
        ${CMAKE_SOURCE_DIR}/source/libs/paragui/include
    )
endif()

# Platform defines (required by ASC codebase)
target_compile_definitions(mcts PUBLIC
    _UNIX_
    SIZE_T_not_identical_to_INT  # Fix size_t ambiguity on 64-bit systems
)

# Link SDL and sigc++
target_link_libraries(mcts PUBLIC
    ${SDL_LIBRARIES}
    ${SIGC_LIBRARIES}
)
target_compile_options(mcts PUBLIC
    ${SDL_CFLAGS_OTHER}
    ${SIGC_CFLAGS_OTHER}
)

# Compiler flags for MCTS module
# Note: Using relaxed warnings during migration to avoid fixing legacy code issues
target_compile_options(mcts PRIVATE
    -Wall
    -Wno-deprecated
    -Wno-unused-parameter
    -fpermissive  # Allow legacy code patterns during migration
)

# Display what we're building
list(LENGTH DOMAIN_SOURCES DOMAIN_COUNT)
list(LENGTH DOMAIN_ABILITY_SOURCES ABILITY_COUNT)
list(LENGTH AGENT_SOURCES AGENT_COUNT)
list(LENGTH CORE_SOURCES CORE_COUNT)
list(LENGTH INFRA_SOURCES INFRA_COUNT)
math(EXPR TOTAL_SOURCES "${DOMAIN_COUNT} + ${ABILITY_COUNT} + ${AGENT_COUNT} + ${CORE_COUNT} + ${INFRA_COUNT}")

message(STATUS "  Library sources: ${TOTAL_SOURCES} files")
message(STATUS "    - Domain: ${DOMAIN_COUNT} files")
message(STATUS "    - Abilities: ${ABILITY_COUNT} files")
message(STATUS "    - Agents: ${AGENT_COUNT} files")
message(STATUS "    - Core: ${CORE_COUNT} files")
message(STATUS "    - Infrastructure: ${INFRA_COUNT} files")

# Test executables
# Note: These were deleted according to git status, but we define them
# in case they're recreated or exist in different branches

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/domain/snapshot_test.cpp")
    add_executable(snapshot_test domain/snapshot_test.cpp)
    target_link_libraries(snapshot_test PRIVATE mcts)
    message(STATUS "  Test: snapshot_test")
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/domain/action_executor_test.cpp")
    add_executable(action_executor_test domain/action_executor_test.cpp)
    target_link_libraries(action_executor_test PRIVATE mcts)
    message(STATUS "  Test: action_executor_test")
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/domain/evaluator_test.cpp")
    add_executable(evaluator_test domain/evaluator_test.cpp)
    target_link_libraries(evaluator_test PRIVATE mcts)
    message(STATUS "  Test: evaluator_test")
endif()

# Enable testing framework
enable_testing()

# Register tests if they were built
if(TARGET snapshot_test)
    add_test(NAME snapshot_test COMMAND snapshot_test)
endif()

if(TARGET action_executor_test)
    add_test(NAME action_executor_test COMMAND action_executor_test)
endif()

if(TARGET evaluator_test)
    add_test(NAME evaluator_test COMMAND evaluator_test)
endif()

message(STATUS "MCTS module configured successfully")
